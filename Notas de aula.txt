git config user.name -> recupera o nome do usuario
git config user.email -> recupera o email do usuario

git config --list -> listar todas as propriedadese E mostrar o user/email

-- Definindo usuario e email
git config --global user.name "nome usuario"
git config --global user.email aluno@email.com.br

-- Apagando usuario e email
git config --global unset user.email
git config --global unset user.name


-- Criando um repo no git
git init

- Obter informação de arquivos e do repositório
git status

-- Selecionar arquivos para entrar no commit
git add nome_arquivo.extesão
git add . -> selecionar todos os arquivos

Untracked -> Added

-- Realizar o commit
git commit -m "a mensagem a ser informada no commit"

-- historico de commits
git log

-- exibir informações de um commit
git show id_commit

-- Criando uma branch principal
git branch -M nome_branch

-- Definindo a origem do repositório
git remote add origin https://link.com.br 

-- Na primeira vez ao executar esse comando, será solicitado a autenticação do GitHub
-- Logo depois é realizado o push ao repo da origem
git push -u origin main

-- Listando branchs
git branch

-- Criando Branch
git checkout -b nome-branch

-- Selecionando branch
git checkout nome-branch
